def get_active_topic_summary(record):
    active_topic_id = record["active_topic_id"]

    for topic in record["candidate_topics"]:
        if topic["topic_id"] == active_topic_id:
            return topic["topic_summary"]

    return None


def format_recent_history(history, n_messages=4):
    recent = history[-n_messages:]

    parts = []

    for message in recent:
        role = message["role"].capitalize()
        content = message["content"]
        parts.append(f"{role}: {content}")

    return "\n".join(parts)


def build_model_input(record):
    recent_context = format_recent_history(record["history"])
    topic_summary = get_active_topic_summary(record)
    current_message = record["current_message"]

    return (
        f"[CLS] "
        f"{recent_context} "
        f"[SEP] "
        f"{topic_summary} "
        f"[SEP] "
        f"{current_message} "
        f"[SEP]"
    )

from sklearn.metrics import accuracy_score, f1_score, precision_recall_fscore_support

def compute_metrics(eval_pred):
    logits, labels = eval_pred
    predictions = np.argmax(logits, axis=-1)

    accuracy = accuracy_score(labels, predictions)

    macro_f1 = f1_score(
        labels,
        predictions,
        average="macro"
    )

    weighted_f1 = f1_score(
        labels,
        predictions,
        average="weighted"
    )

    return {
        "accuracy": accuracy,
        "macro_f1": macro_f1,
        "weighted_f1": weighted_f1
    }

def predict_contextflow(history, active_topic_summary, current_message):
    context = format_recent_history(
        history,
        n_messages=4
    )

    text = (
        f"{context} "
        f"[SEP] "
        f"{active_topic_summary} "
        f"[SEP] "
        f"{current_message}"
    )

    encoded = tokenizer(
        text,
        return_tensors="pt",
        truncation=True,
        max_length=256
    )

    device = next(trainer.model.parameters()).device
    encoded = {k: v.to(device) for k, v in encoded.items()}

    trainer.model.eval()

    with torch.no_grad():
        outputs = trainer.model(**encoded)

    probabilities = torch.softmax(outputs.logits, dim=-1)
    predicted_id = torch.argmax(probabilities, dim=-1).item()

    return {
        "prediction": id2label[predicted_id],
        "confidence": probabilities[0, predicted_id].item(),
        "probabilities": {
            id2label[i]: probabilities[0, i].item()
            for i in range(3)
        }
    }

### Semantic Similarity & Cosine Matchers
We implement cosine similarity search over our topic memory to evaluate how closely a new user message relates to currently stored topics.

import torch
import torch.nn.functional as F

def get_topic_similarities(current_message):
    current_embedding = embedding_model.encode(
        current_message,
        convert_to_tensor=True,
        normalize_embeddings=True
    )

def get_return_evidence(current_message):
    similarities = get_topic_similarities(current_message)

    active_similarity = similarities[active_topic_id]

    previous_similarities = {
        topic_id: score
        for topic_id, score in similarities.items()
        if topic_id != active_topic_id
    }

    best_previous_topic = max(
        previous_similarities,
        key=previous_similarities.get
    )

    best_previous_similarity = previous_similarities[best_previous_topic]

    similarity_margin = (
        best_previous_similarity - active_similarity
    )

    return {
        "active_topic": active_topic_id,
        "active_similarity": active_similarity,
        "best_previous_topic": best_previous_topic,
        "best_previous_similarity": best_previous_similarity,
        "similarity_margin": similarity_margin,
        "all_similarities": similarities
    }

def predict_contextflow(
    history,
    active_topic_summary,
    current_message
):
    context = format_recent_history(
        history,
        n_messages=4
    )

    text = (
        f"{context} "
        f"[SEP] "
        f"{active_topic_summary} "
        f"[SEP] "
        f"{current_message}"
    )

    inputs = tokenizer(
        text,
        return_tensors="pt",
        truncation=True,
        max_length=256
    )

    inputs = {
        key: value.to(device)
        for key, value in inputs.items()
    }

    model.eval()

    with torch.no_grad():
        outputs = model(**inputs)

    probabilities = torch.softmax(
        outputs.logits,
        dim=-1
    )[0]

    predicted_id = torch.argmax(probabilities).item()

    return {
        "prediction": id2label[predicted_id],
        "confidence": probabilities[predicted_id].item(),
        "probabilities": {
            id2label[i]: probabilities[i].item()
            for i in range(3)
        }
    }

def get_hybrid_evidence(

    # Semantic RETURN evidence

## # Context-Aware Semantic Search
We enhance our retrieval accuracy by building a richer context for each candidate topic. Instead of matching just the brief summary, we append prior dialog history to the topic representations.

def build_topic_contexts(record):
    """
    Build a richer text representation for each topic
    using the conversation history associated with that topic.
    """

    topic_contexts = {}

    for topic in record["candidate_topics"]:
        topic_id = topic["topic_id"]
        topic_summary = topic["topic_summary"]

        # Start with the topic summary
        context_parts = [topic_summary]

        # Add conversation history
        for message in record["history"]:
            context_parts.append(
                f"{message['role'].capitalize()}: {message['content']}"
            )

        topic_contexts[topic_id] = " ".join(context_parts)

    return topic_contexts

def get_context_aware_similarities(record):

    current_message = record["current_message"]

    topic_contexts = build_topic_contexts(record)

    current_embedding = embedding_model.encode(
        current_message,
        convert_to_tensor=True,
        normalize_embeddings=True
    )

    similarities = {}

    for topic_id, topic_text in topic_contexts.items():

        topic_embedding = embedding_model.encode(
            topic_text,
            convert_to_tensor=True,
            normalize_embeddings=True
        )

        similarity = F.cosine_similarity(
            current_embedding.unsqueeze(0),
            topic_embedding.unsqueeze(0)
        ).item()

        similarities[topic_id] = similarity

    return similarities
